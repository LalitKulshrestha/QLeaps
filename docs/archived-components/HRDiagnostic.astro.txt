<script>
  function setupDiagnostic() {
    const engine = document.getElementById('diagnosticEngine');
    if (!engine) return;

    const scoreVal = document.getElementById('diagScoreVal');
    const gaugeFill = document.getElementById('diagGaugeFill');
    const statusBadge = document.getElementById('diagStatusBadge');
    const findingsTitle = document.getElementById('diagFindingsTitle');
    const findingsList = document.getElementById('diagFindingsList');
    const actionBtn = document.getElementById('diagActionBtn');

    function calculate() {
      const team = Number(
        engine.querySelector('input[name="diag-team"]:checked')?.value || 10,
      );
      const hr = Number(
        engine.querySelector('input[name="diag-hr"]:checked')?.value || 5,
      );
      const posh = Number(
        engine.querySelector('input[name="diag-posh"]:checked')?.value || 0,
      );
      const hiring = Number(
        engine.querySelector('input[name="diag-hiring"]:checked')?.value || 5,
      );

      const total = Math.min(100, Math.max(10, team + hr + posh + hiring));

      if (scoreVal) scoreVal.textContent = String(total);
      if (gaugeFill) gaugeFill.style.width = `${total}%`;

      let badgeClass = 'status-red';
      let badgeText = 'Critical Exposure';
      let title = 'Immediate Statutory & Operational Gaps';
      let serviceTarget = 'people-setup';
      let items = [];

      if (total < 40) {
        badgeClass = 'status-red';
        badgeText = 'Critical Exposure';
        title = 'High Statutory & Notice Attrition Exposure';
        serviceTarget = 'people-setup';
        items = [
          'Immediate liability: Missing POSH Internal Committee with an external advocate creates direct exposure under Act 2013.',
          'Operational drag: Ad-hoc policies without standardized employment agreements invite dispute and onboarding confusion.',
          'Notice period vulnerability: Unstructured vetting results in high offer dropouts during 60–90 day waiting periods.',
        ];
      } else if (total < 75) {
        badgeClass = 'status-amber';
        badgeText = 'Developing Infrastructure';
        title = 'Operational Gaps at Current Scale';
        serviceTarget = 'people-setup';
        items = [
          'Headcount growth is outpacing handbook policies: Induction, appraisals, and leave systems require codification.',
          'POSH governance needs regularized external training and mandated annual filings to maintain legal standing.',
          'Hiring interviews lack standardized scorecards, leading to subjectivity in engineering and leadership selection.',
        ];
      } else {
        badgeClass = 'status-emerald';
        badgeText = 'Enterprise Scale Ready';
        title = 'Institutional Readiness Established';
        serviceTarget = 'recruitment';
        items = [
          'Foundational people governance is secure and aligned with statutory guidelines.',
          'Primary growth bottleneck: Attracting top senior engineering, design, and marketing leadership.',
          'Recommended focus: Calibrated search dossiers and executive founder thought positioning.',
        ];
      }

      if (statusBadge) {
        statusBadge.className = `score-status-pill ${badgeClass}`;
        statusBadge.textContent = badgeText;
      }
      if (findingsTitle) {
        findingsTitle.textContent = title;
      }
      if (findingsList) {
        findingsList.innerHTML = items.map((i) => `<li>${i}</li>`).join('');
      }
      if (actionBtn) {
        actionBtn.setAttribute(
          'href',
          `/contact/?service=${serviceTarget}&source=diagnostic&score=${total}`,
        );
      }
    }

    engine.querySelectorAll('input[type="radio"]').forEach((input) => {
      input.addEventListener('change', calculate);
    });

    calculate();
  }

  setupDiagnostic();
  document.addEventListener('astro:page-load', setupDiagnostic);
</script>
